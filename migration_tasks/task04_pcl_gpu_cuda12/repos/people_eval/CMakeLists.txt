cmake_minimum_required(VERSION 3.16)
project(people_eval LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(PCL 1.15 REQUIRED COMPONENTS common io gpu_containers gpu_people)
find_package(CUDAToolkit REQUIRED)
add_executable(people_eval src/main.cpp)
target_include_directories(people_eval PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(people_eval PRIVATE ${PCL_LIBRARIES} CUDA::cudart)
install(TARGETS people_eval RUNTIME DESTINATION bin)
