cmake_minimum_required(VERSION 3.16)
project(people_eval LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# VTK 9.1 (pulled in by PCL's config) expects the MPI targets to exist on some distributions
find_package(MPI QUIET COMPONENTS C)
find_package(PCL 1.15 REQUIRED COMPONENTS common io gpu_containers gpu_people)
find_package(CUDAToolkit REQUIRED)
add_executable(people_eval src/main.cpp)
target_include_directories(people_eval PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(people_eval PRIVATE ${PCL_LIBRARIES} CUDA::cudart)
install(TARGETS people_eval RUNTIME DESTINATION bin)
add_executable(ncv_check src/ncv_check.cpp)
target_include_directories(ncv_check PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(ncv_check PRIVATE ${PCL_LIBRARIES} CUDA::cudart)
install(TARGETS ncv_check RUNTIME DESTINATION bin)
