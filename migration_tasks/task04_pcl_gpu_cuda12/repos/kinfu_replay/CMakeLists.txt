cmake_minimum_required(VERSION 3.16)
project(kinfu_replay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(PCL 1.15 REQUIRED COMPONENTS common gpu_containers gpu_kinfu gpu_kinfu_large_scale)
find_package(CUDAToolkit REQUIRED)
find_package(OpenMP REQUIRED)
add_executable(kinfu_replay src/main.cpp src/scene.cpp)
target_include_directories(kinfu_replay PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(kinfu_replay PRIVATE ${PCL_LIBRARIES} CUDA::cudart OpenMP::OpenMP_CXX)
install(TARGETS kinfu_replay RUNTIME DESTINATION bin)
